{{- $ns := include "kopiur-pvc.namespace" . -}}
{{- $items := fromYaml (include "kopiur-pvc.items" .) }}
kopiur-pvc release "{{ .Release.Name }}" installed in namespace {{ $ns }}:
{{- range $pvcName, $override := $items }}
{{- $override = $override | default dict -}}
{{- $pvcCfg := mergeOverwrite (deepCopy $.Values.pvc) ($override.pvc | default dict) -}}
{{- $populateCfg := mergeOverwrite (deepCopy $.Values.populate) ($override.populate | default dict) -}}
{{- $backupCfg := mergeOverwrite (deepCopy $.Values.backup) ($override.backup | default dict) }}

{{ $pvcName }}:
  - PersistentVolumeClaim/{{ $pvcName }} ({{ $pvcCfg.size }}{{ if $pvcCfg.storageClassName }}, storageClass={{ $pvcCfg.storageClassName }}{{ end }})
{{- if $populateCfg.enabled }}
  - Restore/{{ $pvcName }} (passive populator, onMissingSnapshot={{ $populateCfg.onMissingSnapshot }})
{{- end }}
{{- if $backupCfg.enabled }}
  - SnapshotPolicy/{{ $pvcName }} -> {{ $backupCfg.repository.kind }}/{{ $backupCfg.repository.name }} (copyMethod={{ $backupCfg.copyMethod }})
{{- if $backupCfg.schedule.cron }}
  - SnapshotSchedule/{{ $pvcName }} (cron="{{ $backupCfg.schedule.cron }}")
{{- else }}
  - No SnapshotSchedule rendered (schedule.cron is empty) — trigger backups manually or with your own SnapshotSchedule/policySelector.
{{- end }}
{{- else }}
  - backup.enabled=false — no SnapshotPolicy/SnapshotSchedule created; this PVC is not backed up.
{{- end }}
{{- end }}
